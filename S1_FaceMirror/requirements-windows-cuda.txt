# S1 FaceMirror — Windows + CUDA 12.1 install requirements
#
# Install with:
#   python -m pip install --upgrade pip
#   pip install -r requirements-windows-cuda.txt
#
# The --extra-index-url at the top routes torch to PyTorch's CUDA 12.1 wheels.
# All other packages resolve from PyPI as normal.
#
# Prerequisites on the build/runtime machine:
#   - Python 3.10 (matches the macOS lockfile; pyfaceau requires >=3.10)
#   - NVIDIA driver supporting CUDA 12.1 (driver R530+ recommended)
#   - Microsoft Visual C++ Build Tools 2022 (only required for installing
#     pyfaceau from sdist; not needed at runtime once the PyInstaller bundle
#     is built)
#
# Versions are intentionally loose. Lock with `pip freeze` on a known-good
# Windows machine and commit as requirements-windows-cuda.lock when ready.

--extra-index-url https://download.pytorch.org/whl/cu121

# --- Core CUDA-aware stack ---------------------------------------------------
# torch: CUDA 12.1 wheel (filename will be torch-X.Y.Z+cu121-cp310-cp310-win_amd64.whl)
torch>=2.5.0

# ONNX Runtime with CUDAExecutionProvider. pymtcnn and pyfaceau both pick this
# up automatically when they detect CUDA is available.
onnxruntime-gpu>=1.18.0

# --- FaceMirror dependency stack (PyPI) --------------------------------------
# All four packages have working Windows installs:
#   pyfhog   — has cp310-win_amd64 wheels
#   pyclnf   — pure-Python wheel; uses torch for GPU work
#   pymtcnn  — pure-Python wheel; lazy-imports onnxruntime CUDA backend
#   pyfaceau — sdist-only; compiles Cython exts via MSVC, falls back to pure
#              Python if MSVC unavailable (slower but functional)
pyfhog>=0.1.4
pyclnf>=0.3.3
pymtcnn>=1.1.5
pyfaceau>=1.3.11

# --- Vision / numerics -------------------------------------------------------
